# 反调试与反逆向技术对抗

## 1. 概述

反调试（Anti-Debugging）与反逆向（Anti-RE）技术是目标程序为了保护自身不被分析的防御手段。在CTF竞赛中，中高难度的逆向题目普遍包含反调试措施——它们不是题目的核心考点，但却是分析的"拦路虎"。掌握反调试绕过技术的意义在于：花最少的时间越过防御，把精力集中在真正的算法分析上。

反调试技术大致分为五类：
1. **调试器检测**：检测调试器是否附加
2. **环境检测**：检测是否在虚拟机/沙箱中运行
3. **时间检测**：通过时间差检测单步执行
4. **完整性校验**：检测代码是否被修改（软件断点）
5. **主动干扰**：使调试器行为异常或崩溃

## 2. 核心原理与检测手法

### 2.1 调试器存在检测

#### 2.1.1 Windows PEB标志位检测

Windows进程的PEB结构中，`BeingDebugged`（偏移+0x2）、`NtGlobalFlag`（偏移+0x68）标志位泄露调试器状态：

```asm
; x86: PEB检测 BeingDebugged
mov eax, fs:[0x30]      ; 获取PEB地址
movzx eax, byte [eax+2] ; BeingDebugged
test eax, eax
jnz debugger_detected

; x64:
mov rax, gs:[0x60]      ; x64 PEB在GS段
movzx eax, byte [rax+2]
test eax, eax
jnz debugger_detected

; NtGlobalFlag检测
mov eax, fs:[0x30]
mov eax, [eax+0x68]     ; NtGlobalFlag
and eax, 0x70           ; FLG_HEAP_ENABLE_TAIL_CHECK |
                         ; FLG_HEAP_ENABLE_FREE_CHECK |
                         ; FLG_HEAP_VALIDATE_PARAMETERS
jnz debugger_detected
```

**绕过方法**：
- x64dbg + ScyllaHide插件（自动修改PEB）
- 手动修改：调试时在对应位置写入0
- IDA Patch反汇编：NOP掉检测逻辑

#### 2.1.2 API调用检测

```cpp
// IsDebuggerPresent() - 最基础的调试器检测
if (IsDebuggerPresent()) exit(-1);

// CheckRemoteDebuggerPresent()
BOOL debuggerPresent = FALSE;
CheckRemoteDebuggerPresent(GetCurrentProcess(), &debuggerPresent);
if (debuggerPresent) exit(-1);

// NtQueryInformationProcess 通过 ProcessDebugPort (0x7)
DWORD debugPort = 0;
NtQueryInformationProcess(
    GetCurrentProcess(),
    ProcessDebugPort,       // 0x7
    &debugPort, sizeof(debugPort), NULL
);
if (debugPort != 0) exit(-1);  // 调试端口非0表示有调试器附加

// NtQueryInformationProcess 通过 ProcessDebugFlags (0x1F)
// 正常进程：DebugFlags = 1; 被调试进程：DebugFlags = 0
DWORD debugFlags = 0;
NtQueryInformationProcess(
    GetCurrentProcess(),
    ProcessDebugFlags,     // 0x1F
    &debugFlags, sizeof(debugFlags), NULL
);
if (debugFlags == 0) exit(-1);
```

#### 2.1.3 进程黑名单检测

扫描进程名或窗口标题，检测已知调试器：

```cpp
// 检测常见调试器窗口
if (FindWindow("OllyDbg", NULL))      exit(-1);
if (FindWindow("WinDbgFrameClass", NULL)) exit(-1);
if (FindWindow("x64dbg", NULL))       exit(-1);

// 通过Toolhelp32枚举进程
HANDLE snap = CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS, 0);
PROCESSENTRY32 pe32;
pe32.dwSize = sizeof(PROCESSENTRY32);
if (Process32First(snap, &pe32)) {
    do {
        if (strstr(pe32.szExeFile, "x64dbg") ||
            strstr(pe32.szExeFile, "ollydbg") ||
            strstr(pe32.szExeFile, "windbg")) {
            exit(-1);
        }
    } while (Process32Next(snap, &pe32));
}
```

### 2.2 硬件断点检测

调试寄存器（DR0-DR3）非零说明可能有硬件断点：

```cpp
// 通过GetThreadContext读取DR寄存器
CONTEXT ctx;
ctx.ContextFlags = CONTEXT_DEBUG_REGISTERS;
GetThreadContext(GetCurrentThread(), &ctx);
if (ctx.Dr0 || ctx.Dr1 || ctx.Dr2 || ctx.Dr3) {
    // 检测到硬件断点
    exit(-1);
}
```

**绕过方法**：使用内存断点代替硬件断点，或ScyllaHide的DRx保护。

### 2.3 软件断点（INT3）扫描

```cpp
// 对关键函数代码段做完整性哈希
DWORD oldProtect;
BYTE* funcStart = (BYTE*)CriticalFunction;
VirtualProtect(funcStart, funcSize, PAGE_EXECUTE_READWRITE, &oldProtect);
DWORD checksum = 0;
for (DWORD i = 0; i < funcSize; i++) {
    checksum = (checksum + funcStart[i]) ^ 0x12345678;
}
VirtualProtect(funcStart, funcSize, oldProtect, &oldProtect);
if (checksum != EXPECTED_CHECKSUM) exit(-1);

// 逐字节搜索0xCC
for (DWORD i = 0; i < funcSize; i++) {
    if (funcStart[i] == 0xCC) {
        exit(-1); // 检测到INT3
    }
}
```

**绕过方法**：
- 优先使用硬件断点（不修改代码）
- 在INT3扫描代码之前就触发断点

### 2.4 时间检测

通过比较时间戳检测是否被单步调试：

```cpp
// RDTSC方式：精确到CPU周期
LARGE_INTEGER t1, t2;
t1 = __rdtsc();
// 执行一些不耗时的操作（如 add eax, ebx）
t2 = __rdtsc();
if (t2.QuadPart - t1.QuadPart > THRESHOLD) {
    // 时间差过大，说明中间有调试器停驻
    exit(-1);
}

// QueryPerformanceCounter方式
LARGE_INTEGER freq, start, end;
QueryPerformanceFrequency(&freq);
QueryPerformanceCounter(&start);
// 执行几条指令
QueryPerformanceCounter(&end);
double elapsed = (double)(end.QuadPart - start.QuadPart) / freq.QuadPart;
if (elapsed > 0.001) exit(-1);  // 超过1ms即为异常

// GetTickCount反复比较
DWORD t1 = GetTickCount();
Sleep(100);  // 预期100ms
DWORD t2 = GetTickCount();
if (abs((int)(t2 - t1) - 100) > 10) {
    // 实际sleep时间远超100ms，有调试器停顿
    exit(-1);
}
```

**绕过方法**：
- 在时间检测相关的API/指令处设断点，修改返回值（让t1==t2）
- 使用ScyllaHide的时间反检测功能
- 手动Patch比较逻辑（把阈值阈值改大）

### 2.5 Linux平台反调试

#### 2.5.1 ptrace自检

Linux中同时只能有一个调试器ptrace一个进程。利用这个特性，自己ptrace自己来阻止其他调试器：

```c
// 经典自ptrace反调试
if (ptrace(PTRACE_TRACEME, 0, 0, 0) == -1) {
    // ptrace失败说明已经被调试
    exit(-1);
}

// 检测/proc/self/status中的TracerPid
int fd = open("/proc/self/status", O_RDONLY);
char buf[1024];
read(fd, buf, sizeof(buf));
char* tracer = strstr(buf, "TracerPid:");
int pid = atoi(tracer + 10);
if (pid != 0) {
    exit(-1);  // 有tracer
}
```

**绕过方法**：
```bash
# gdb中：在ptrace调用处patch返回值
# 断点位置：b *main+0xXX（ptrace调用的地址）
b *0x401234
commands
  silent
  set $rax = 0          # 伪造返回值为0（成功）
  continue
end

# 或者用LD_PRELOAD劫持ptrace
cat > antiptrace.c << 'EOF'
long ptrace(int request, ...) { return 0; }
EOF
gcc -shared -fPIC antiptrace.c -o antiptrace.so
LD_PRELOAD=./antiptrace.so ./binary
```

#### 2.5.2 子进程检测

```c
// fork一个子进程，子进程调试父进程
pid_t child = fork();
if (child == 0) {
    // 子进程：尝试attach父进程
    ptrace(PTRACE_ATTACH, getppid(), 0, 0);
    // 如果attach成功，父进程已被调试
    // 如果attach失败，父进程正常
    exit(0);
} else {
    // 父进程：继续正常逻辑
    waitpid(child, NULL, 0);
}
```

## 3. 关键绕过工具与技巧

### 3.1 ScyllaHide配置

ScyllaHide是x64dbg和OllyDbg的王牌反反调试插件：

```
# ScyllaHide.ini 关键配置
[x64dbg]
; 勾选以下Profile以适应不同目标：
; - Basic: 修改PEB、NtGlobalFlag
; - Anti-Anti-Attach: 防止进程检测调试器存在
; - PE Header: 清除PE头中的调试标志
; - NtQuerySystemInformation: Hook系统信息API
; - NtSetInformationThread: 防止被隐藏线程
; - NtQueryInformationProcess: 拦截各种Process查询
; - NtYieldExecution: 防止反调试线程
; - Hook RDTSC: 让RDTSC返回恒定值（对抗时间检测）
```

### 3.2 IDA脚本化反调试绕过

```python
# 一键NOP所有反调试检测函数
import idaapi
import idautils
import idc

ANTI_DEBUG_APIS = [
    "IsDebuggerPresent",
    "CheckRemoteDebuggerPresent",
    "NtQueryInformationProcess",
    "OutputDebugStringA",  # OutputDebugString + SetLastError组合检测
    "NtSetInformationThread",
    "FindWindowA",
    "FindWindowW",
    "GetTickCount",
    "QueryPerformanceCounter",
    "RtlGetNtGlobalFlags",
]

def patch_anti_debug_calls():
    patched = 0
    for api_name in ANTI_DEBUG_APIS:
        api_ea = idc.get_name_ea_simple(api_name)
        if api_ea == idaapi.BADADDR:
            continue
        for xref in idautils.CodeRefsTo(api_ea, False):
            # 获取调用指令的地址
            call_ea = xref
            call_len = idc.get_item_size(call_ea)
            
            # 用NOP替换call指令
            for i in range(call_len):
                idc.patch_byte(call_ea + i, 0x90)
            
            # 如果返回值被检测，可能需要进一步放在返回值eax/rax
            idc.set_cmt(call_ea, "PATCHED: anti-debug call removed", False)
            patched += 1
    print(f"[*] Patched {patched} anti-debug calls")

patch_anti_debug_calls()
```

### 3.3 TLS回调检测与绕过

TLS（Thread Local Storage）回调在main函数之前执行，常被用于隐藏反调试初始化：

```python
# 在IDA中找到TLS回调
# PE头 -> IMAGE_DATA_DIRECTORY[9] -> TLS Directory
# TLS Directory结构中的AddressOfCallBacks指向回调函数数组

import idc
import struct

def find_tls_callbacks():
    pe_offset = idc.get_wide_dword(idc.get_imagebase() + 0x3C)
    # DataDirectory[9] TLS
    tls_rva = idc.get_wide_dword(
        idc.get_imagebase() + pe_offset + 0x18 + 9*8
    )
    tls_va = idc.get_imagebase() + tls_rva
    # TLS callback array address
    callback_rva = idc.get_wide_dword(tls_va + 0xC)
    callback_va = idc.get_imagebase() + callback_rva
    
    i = 0
    while True:
        cb = idc.get_wide_dword(callback_va + i*4)
        if cb == 0:
            break
        print(f"[*] TLS Callback {i}: {hex(idc.get_imagebase() + cb)}")
        i += 1

find_tls_callbacks()
```

## 4. 常见误区与注意事项

- **误区**：认为Patch掉检测代码就万事大吉。有些程序有多处冗余检测，Patch一处后其他位置仍会触发。务必搜索所有检测API的交叉引用。
- **误区**：忘记了反调试只是"入场券"。绕过反调试后真正的挑战在于逆向算法本身，不要花太多时间纠结防御绕过。
- **注意**：现代反调试会使用异常处理（SEH/VEH）来隐藏检测代码，IDA可能无法正常显示。需要在反编译视图中仔细查找 `__try/__except` 块。
- **注意**：某些反调试手法嵌套使用——例如用TLS回调修改main函数的代码，使得直接Patch失效。
- **安全**：绕过反调试时，务必在受控环境中操作（虚拟机快照），避免运行恶意负载。

## 5. 实战示例

### 场景：一个多层反调试保护的CTF题目

```
第一层：IsDebuggerPresent + CheckRemoteDebuggerPresent
→ ScyllaHide自动绕过了

第二层：TLS回调中检测PEB.NtGlobalFlag
→ ScyllaHide的PEB保护也处理了

第三层：自定义的异常处理检测
  __try { 
    __asm { int 3 } 
  } __except(filter) {
    // 正常执行时filter返回EXCEPTION_EXECUTE_HANDLER
    // 调试器下INT3被调试器捕获，不会走到这里
    // 因此程序会意识到被调试
  }
→ 手动方法：在filter函数返回前强制设置返回值为EXCEPTION_EXECUTE_HANDLER

第四层：时间检测
  RDTSC两次比较
→ x64dbg断在RDTSC之后，修改第二次RDTSC的返回值

第五层：CRC校验代码段完整性
  检查关键函数是否被修改
→ 不使用软件断点，全部改用硬件断点
```

**x64dbg操作日志**：
```
1. Alt+E (符号窗口) -> 找到 IsDebuggerPresent 设断点
2. F9运行，断在 IsDebuggerPresent -> Ctrl+F9 (运行到返回)
3. 在返回处修改 eax = 0 (未调试)
4. 继续运行，同样处理 CheckRemoteDebuggerPresent
5. 在 TLS 回调入口设F2，观察是否在main前执行
6. 在程序中搜索 RDTSC (0F 31)，设硬件执行断点
7. 每次RDTSC触发后，统一修改 edx:eax 为相同的值
8. 完成所有绕过后，F9正常运行到核心逻辑
```

> **经验法则**：反调试的本质是代码中的"if检测 - exit"。你的目标不是理解检测的每一个细节，而是找到那个 `exit(-1)` 并让它永远不可达。最快的绕过方法是：找到所有反调试相关的 `call` 指令，要么NOP掉，要么在调用处修改返回值。工具（ScyllaHide）+ 手动Patch结合，效率最高。
