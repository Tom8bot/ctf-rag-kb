## 2. 核心原理

### 2.1 模运算基础

```
定义: a ≡ b (mod m) 表示 a 和 b 除以 m 的余数相同

基本性质:
1. (a + b) mod m = ((a mod m) + (b mod m)) mod m
2. (a - b) mod m = ((a mod m) - (b mod m)) mod m
3. (a × b) mod m = ((a mod m) × (b mod m)) mod m
4. 若 a ≡ b (mod m) 且 c ≡ d (mod m)，则 a+c ≡ b+d (mod m)

模逆元:
  若 a × b ≡ 1 (mod m)，则 b 是 a 在模 m 下的逆元，记为 a⁻¹ mod m
  存在条件: gcd(a, m) = 1
  求法: 扩展欧几里得算法
```

### 2.2 欧拉函数

```
φ(n): 小于 n 且与 n 互质的正整数的个数

性质:
1. 若 p 为素数，则 φ(p) = p - 1
2. 若 p,q 为不同素数，则 φ(pq) = (p-1)(q-1)
3. 若 n = p₁^e₁ × p₂^e₂ × ...，则
   φ(n) = n × (1 - 1/p₁) × (1 - 1/p₂) × ...

计算 φ(60):
  60 = 2² × 3 × 5
  φ(60) = 60 × (1 - 1/2) × (1 - 1/3) × (1 - 1/5)
        = 60 × 1/2 × 2/3 × 4/5
        = 16
```

### 2.3 欧拉定理与RSA

```
欧拉定理: 若 gcd(a, n) = 1，则 a^φ(n) ≡ 1 (mod n)

费马小定理（特殊情况）: 若 p 为素数且 a 不被 p 整除，则
  a^(p-1) ≡ 1 (mod p)

RSA加密:
  密钥生成:
    1. 选大素数 p, q
    2. n = p × q
    3. φ(n) = (p-1)(q-1)
    4. 选 e 使得 gcd(e, φ(n)) = 1 (通常 e = 65537)
    5. 计算 d = e⁻¹ mod φ(n)
  
  公钥: (n, e)
  私钥: (n, d)
  
  加密: c = m^e mod n
  解密: m = c^d mod n
  
  正确性验证:
    c^d ≡ (m^e)^d ≡ m^(ed) ≡ m^(kφ(n)+1) ≡ m (mod n)
    (由欧拉定理: m^φ(n) ≡ 1 mod n)
```

### 2.4 中国剩余定理(CRT)

```
问题: 求解同余方程组
  x ≡ a₁ (mod m₁)
  x ≡ a₂ (mod m₂)
  ...
  x ≡ aₖ (mod mₖ)
其中 m₁, m₂, ..., mₖ 两两互质

解法:
  1. M = m₁ × m₂ × ... × mₖ
  2. 对每个 i, Mᵢ = M / mᵢ
  3. 对每个 i, yᵢ = Mᵢ⁻¹ mod mᵢ
  4. x = Σ(aᵢ × Mᵢ × yᵢ) mod M

示例: x ≡ 2 (mod 3), x ≡ 3 (mod 5), x ≡ 2 (mod 7)
  M = 3 × 5 × 7 = 105
  M₁ = 35, y₁ = 35⁻¹ mod 3 = 2
  M₂ = 21, y₂ = 21⁻¹ mod 5 = 1
  M₃ = 15, y₃ = 15⁻¹ mod 7 = 1
  
  x = (2×35×2 + 3×21×1 + 2×15×1) mod 105
    = (140 + 63 + 30) mod 105
    = 233 mod 105
    = 23
```

```python
#!/usr/bin/env python3
"""
number_theory_tools.py - CTF密码学数论工具集
"""

import gmpy2
from Crypto.Util.number import *
import math
from typing import Tuple, List, Optional

class NumberTheory:
    """数论工具类"""
    
    # ===== 基础运算 =====
    @staticmethod
    def gcd(a: int, b: int) -> int:
        """最大公约数 (欧几里得算法)"""
        return math.gcd(a, b)  # 或用 gmpy2.gcd(a, b)
    
    @staticmethod
    def lcm(a: int, b: int) -> int:
        """最小公倍数"""
        return abs(a * b) // math.gcd(a, b)
    
    @staticmethod
    def egcd(a: int, b: int) -> Tuple[int, int, int]:
        """扩展欧几里得算法
        返回 (g, x, y) 使得 a*x + b*y = g = gcd(a,b)
        """
        if b == 0:
            return (a, 1, 0)
        g, x1, y1 = NumberTheory.egcd(b, a % b)
        x = y1
        y = x1 - (a // b) * y1
        return (g, x, y)
    
    @staticmethod
    def modinv(a: int, m: int) -> Optional[int]:
        """模逆元: a^(-1) mod m"""
        g, x, _ = NumberTheory.egcd(a, m)
        if g != 1:
            return None  # 不存在
        return x % m
    
    @staticmethod
    def modinv_gmpy2(a: int, m: int) -> int:
        """模逆元 (gmpy2版本, 更高性能)"""
        return int(gmpy2.invert(a, m))
    
    # ===== 素数相关 =====
    @staticmethod
    def is_prime(n: int) -> bool:
        """素数判断 (Miller-Rabin)"""
        return gmpy2.is_prime(n)
    
    @staticmethod
    def next_prime(n: int) -> int:
        """下一个素数"""
        return int(gmpy2.next_prime(n))
    
    @staticmethod
    def generate_prime(bits: int) -> int:
        """生成指定位数的素数"""
        return int(gmpy2.next_prime(
            int(gmpy2.mpz_random(gmpy2.random_state(), 2**bits))
        ))
    
    @staticmethod
    def factorize(n: int, timeout: int = 30) -> dict:
        """因式分解 (使用多种方法)"""
        import time
        start = time.time()
        
        # 方法1: 试除法 (小因子)
        for p in range(2, min(1000000, int(n**0.5) + 1)):
            if time.time() - start > timeout:
                break
            if n % p == 0:
                factors = {}
                while n % p == 0:
                    factors[p] = factors.get(p, 0) + 1
                    n //= p
        
        # 方法2: Pollard's Rho (中等大小的因子)
        if n > 1:
            try:
                factor = NumberTheory.pollard_rho(n)
                if factor and factor != n:
                    factors = {factor: 1, n // factor: 1}
                    return factors
            except:
                pass
        
        # 方法3: sympy (如果可用)
        try:
            import sympy
            return sympy.factorint(n)
        except ImportError:
            pass
        
        return {n: 1} if n > 1 else {}
    
    @staticmethod
    def pollard_rho(n: int) -> Optional[int]:
        """Pollard's Rho因式分解算法"""
        if n % 2 == 0:
            return 2
        if n % 3 == 0:
            return 3
        
        x = 2
        y = 2
        d = 1
        f = lambda x: (x * x + 1) % n
        
        while d == 1:
            x = f(x)
            y = f(f(y))
            d = math.gcd(abs(x - y), n)
        
        return d if d != n else None
    
    # ===== 离散对数 =====
    @staticmethod
    def baby_step_giant_step(g: int, h: int, p: int) -> Optional[int]:
        """Baby-step Giant-step算法求解离散对数
        求 x 使得 g^x ≡ h (mod p)
        """
        import math
        
        n = int(math.ceil(math.sqrt(p - 1)))
        
        # Baby steps: g^j mod p for j in [0, n)
        table = {}
        value = 1
        for j in range(n):
            table[value] = j
            value = (value * g) % p
        
        # Giant steps: g^(-n) mod p
        g_inv_n = pow(g, -n, p)  # Python 3.8+支持负数指数
        
        # 搜索 h * g^(-i*n)
        y = h
        for i in range(n):
            if y in table:
                return i * n + table[y]
            y = (y * g_inv_n) % p
        
        return None
    
    # ===== 中国剩余定理 =====
    @staticmethod
    def crt(remainders: List[int], moduli: List[int]) -> Tuple[int, int]:
        """中国剩余定理
        返回 (x, M) 其中 x ≡ remainders[i] (mod moduli[i])
        M = ∏ moduli[i]
        """
        M = 1
        for m in moduli:
            M *= m
        
        x = 0
        for a, m in zip(remainders, moduli):
            Mi = M // m
            yi = pow(Mi, -1, m)  # Mi^(-1) mod m
            x = (x + a * Mi * yi) % M
        
        return (x, M)
    
    @staticmethod
    def crt_gmpy2(remainders: List[int], moduli: List[int]) -> int:
        """使用gmpy2的CRT"""
        result, _ = gmpy2.iroot(1, 2)  # 初始化
        # gmpy2 没有直接的CRT函数，使用手动计算
        M = 1
        for m in moduli:
            M *= m
        
        x = gmpy2.mpz(0)
        for a, m in zip(remainders, moduli):
            Mi = M // m
            yi = gmpy2.invert(gmpy2.mpz(Mi), gmpy2.mpz(m))
            x = (x + a * Mi * yi) % M
        
        return int(x)
    
    # ===== 二次剩余 =====
    @staticmethod
    def legendre_symbol(a: int, p: int) -> int:
        """Legendre符号 (a/p)
        返回: 0(整除), 1(平方剩余), -1(非剩余)
        """
        if a % p == 0:
            return 0
        return pow(a, (p - 1) // 2, p) if pow(a, (p - 1) // 2, p) == 1 else -1
    
    @staticmethod
    def tonelli_shanks(n: int, p: int) -> Optional[int]:
        """Tonelli-Shanks算法求解平方剩余
        求 x 使得 x^2 ≡ n (mod p)，其中p是奇素数
        """
        if NumberTheory.legendre_symbol(n, p) != 1:
            return None  # 无解
        
        # 特殊情况
        if p % 4 == 3:
            return pow(n, (p + 1) // 4, p)
        
        # Tonelli-Shanks
        Q = p - 1
        S = 0
        while Q % 2 == 0:
            Q //= 2
            S += 1
        
        # 找到二次非剩余
        z = 2
        while NumberTheory.legendre_symbol(z, p) != -1:
            z += 1
        
        M = S
        c = pow(z, Q, p)
        t = pow(n, Q, p)
        R = pow(n, (Q + 1) // 2, p)
        
        while t != 1:
            # 找到最小的 i 使得 t^(2^i) = 1
            i = 1
            temp = (t * t) % p
            while temp != 1:
                temp = (temp * temp) % p
                i += 1
            
            b = pow(c, 1 << (M - i - 1), p)
            M = i
            c = (b * b) % p
            t = (t * c) % p
            R = (R * b) % p
        
        return R
    
    # ===== Wiener攻击辅助 =====
    @staticmethod
    def continued_fraction(num: int, den: int) -> List[int]:
        """计算连分数展开"""
        cf = []
        while den:
            q, r = divmod(num, den)
            cf.append(q)
            num, den = den, r
        return cf
    
    @staticmethod
    def convergents_from_cf(cf: List[int]) -> List[Tuple[int, int]]:
        """从连分数计算收敛分数"""
        convs = []
        for i in range(len(cf)):
            if i == 0:
                n, d = cf[0], 1
            elif i == 1:
                n, d = cf[1] * cf[0] + 1, cf[1]
            else:
                n = cf[i] * convs[-1][0] + convs[-2][0]
                d = cf[i] * convs[-1][1] + convs[-2][1]
            convs.append((n, d))
        return convs


# ===== 常用攻击函数 =====
class RSAttacks:
    """RSA常见攻击集合"""
    
    @staticmethod
    def small_e_attack(e: int, c: int, n: int) -> Optional[bytes]:
        """小指数攻击 (e很小，c = m^e < n)"""
        m, exact = gmpy2.iroot(c, e)
        if exact:
            return long_to_bytes(int(m))
        return None
    
    @staticmethod
    def common_modulus_attack(n, e1, e2, c1, c2):
        """共模攻击"""
        g, s1, s2 = gmpy2.gcdext(e1, e2)
        if g != 1:
            return None
        if s1 < 0:
            c1 = int(gmpy2.invert(c1, n))
            s1 = -s1
        if s2 < 0:
            c2 = int(gmpy2.invert(c2, n))
            s2 = -s2
        m = (pow(c1, s1, n) * pow(c2, s2, n)) % n
        return long_to_bytes(m)
    
    @staticmethod
    def wiener_attack(n: int, e: int, c: int) -> Optional[bytes]:
        """Wiener攻击 (私钥d很小)"""
        nt = NumberTheory()
        cf = nt.continued_fraction(e, n)
        convs = nt.convergents_from_cf(cf)
        
        for k, d in convs:
            if k == 0:
                continue
            
            # 检查 phi_n = (e*d - 1) / k 是否为整数
            if (e * d - 1) % k != 0:
                continue
            
            phi_n = (e * d - 1) // k
            
            # 验证: 通过二次方程 x^2 - (n-phi_n+1)x + n = 0
            # 解应该为 p 和 q
            b = n - phi_n + 1
            discriminant = b * b - 4 * n
            
            if discriminant >= 0:
                sqrt_disc, exact = gmpy2.iroot(discriminant, 2)
                if exact:
                    p = (b + int(sqrt_disc)) // 2
                    q = (b - int(sqrt_disc)) // 2
                    if p * q == n:
                        m = pow(c, d, n)
                        return long_to_bytes(m)
        
        return None
    
    @staticmethod
    def hastad_broadcast(n_list, e, c_list):
        """Hastad广播攻击"""
        nt = NumberTheory()
        
        # 使用CRT组合密文
        x, _ = nt.crt(c_list, n_list)
        
        # 求e次根
        m, exact = gmpy2.iroot(x, e)
        if exact:
            return long_to_bytes(int(m))
        return None
    
    @staticmethod
    def franklin_reiter(n, e, c1, c2, r):
        """Franklin-Reiter相关消息攻击
        m2 = m1 + r (已知r)
        """
        # 使用多项式GCD
        P.<x> = PolynomialRing(Zmod(n))
        
        f1 = x^e - c1
        f2 = (x + r)^e - c2
        
        # 计算gcd
        g = gcd(f1, f2)
        
        # 如果gcd是线性的，根即为消息
        if g.degree() == 1:
            root = -g[0] / g[1]
            return long_to_bytes(int(root))
        return None
```

### 3.2 常用攻击数学条件速查

```python
"""
CTF中RSA攻击的数学条件速查表

攻击类型            | 前提条件                          | 数学原理
-------------------|----------------------------------|----------
小指数攻击          | m^e < n (没有取模效果)             | 直接在整数范围内开e次根
共模攻击            | 同一m用不同e加密，模n相同           | 扩展欧几里得 s1*e1 + s2*e2 = 1
Wiener攻击          | d < n^(1/4)/3 (d很小)             | 连分数近似 e/n ≈ k/d
Hastad广播攻击      | 同一m被e个不同n加密，e相同          | CRT + 开e次根
Fermat因式分解      | |p-q| 很小                        | n = ((p+q)/2)^2 - ((p-q)/2)^2
Pollard p-1        | p-1是平滑的(小素因子的乘积)         | Fermat小定理
Boneh-Durfee       | d < n^0.292 (比Wiener更宽松)      | 格归约(LLL算法)
Coppersmith        | 知道部分明文信息                    | 格归约+小根求解
Franklin-Reiter    | m1和m2有已知线性关系               | 多项式GCD
"""

# 条件判断函数
def diagnose_rsa_attack(n, e, c_list=None, n_list=None):
    """诊断适合的RSA攻击类型"""
    
    print(f"[*] n bits: {n.bit_length()}")
    print(f"[*] e = {e}")
    
    suggestions = []
    
    # 检查1: 小指数
    if e <= 17:
        suggestions.append("小指数攻击: e很小，尝试直接开e次根")
    
    # 检查2: Wiener攻击
    if e > n // 3:  # 粗略判断: e很大意味着d很小
        suggestions.append("Wiener攻击: e很大，d可能很小")
    
    # 检查3: Fermat因式分解
    # 需要实际尝试，这里只给出建议
    suggestions.append("Fermat因式分解: 如果p和q很接近")
    
    # 检查4: Pollard p-1
    suggestions.append("Pollard p-1: 如果p-1或q-1是平滑的")
    
    # 检查5: 共模攻击
    if c_list and n_list and len(c_list) >= 2:
        if all(n_val == n for n_val in n_list):
            suggestions.append("共模攻击: 检测到多个密文使用相同的n")
    
    # 检查6: Hastad广播攻击
    if c_list and n_list and len(c_list) >= e:
        suggestions.append("Hastad广播攻击: 密文数量 >= e")
    
    print("\n[*] 建议尝试的攻击:")
    for s in suggestions:
        print(f"  - {s}")
    
    return suggestions
```

```python
#!/usr/bin/env python3

    print(f'{conv.p}/{conv.q} = {float(conv.p/conv.q):.10f}')
```

## 4. 常见误区与注意事项

### 4.1 Crypto解题常见误区

| 误区 | 正确做法 |
|------|----------|
| 只关注n是否可分解 | 先检查所有数学条件（e大小、多密文、padding问题） |
| 认为大素数无法分解 | 试试Fermat、Pollard p-1、William p+1等方法 |
| 忽视编码问题 | 密文和明文之间可能有多层编码 |
| 忽略明文的结构特征 | 明文可能有固定前缀（flag{）或padding模式 |
| 分不清RSA的padding | PKCS1_v1.5和OAEP的padding结构完全不同 |
| 忘记检查e和φ(n)是否互质 | gcd(e, φ(n)) 必须为1，否则解密不是唯一 |
| 使用标准库但参数不对 | AES的模式、IV、padding必须完全匹配 |

### 示例1：完整RSA多策略攻击流程

```python
#!/usr/bin/env python3
"""RSA多策略自动攻击"""

from Crypto.Util.number import *
import gmpy2
import sympy
from number_theory_tools import NumberTheory, RSAttacks

def auto_rsa_attack(n, e, c, context=None):
    """自动尝试RSA常见攻击"""
    
    print(f"[*] 开始RSA攻击分析")
    print(f"    n bits = {n.bit_length()}")
    print(f"    e = {e}")
    
    attacker = RSAttacks()
    
    # 攻击1: 小指数
    if e <= 17:
        print("[*] 尝试小指数攻击...")
        result = attacker.small_e_attack(e, c, n)
        if result:
            print(f"[+] 成功! Flag: {result}")
            return result
        print("[-] 小指数攻击失败")
    
    # 攻击2: 因式分解
    print("[*] 尝试因式分解 n...")
    factors = sympy.factorint(n, limit=1000000)
    if len(factors) > 1 and factors.get(n, 0) != 1:
        print(f"[+] 分解成功: {factors}")
        p = list(factors.keys())[0]
        q = n // p
        phi = (p-1) * (q-1)
        d = int(gmpy2.invert(e, phi))
        m = pow(c, d, n)
        flag = long_to_bytes(m)
        print(f"[+] Flag: {flag}")
        return flag
    print("[-] 因式分解失败")
    
    # 攻击3: Wiener攻击
    if e > n >> 1:  # e很大
        print("[*] 尝试Wiener攻击...")
        result = attacker.wiener_attack(n, e, c)
        if result:
            print(f"[+] 成功! Flag: {result}")
            return result
        print("[-] Wiener攻击失败")
    
    # 攻击4: 尝试Fermat因式分解
    print("[*] 尝试Fermat因式分解...")
    # Fermat: 寻找 a = ceil(sqrt(n)) + i 使得 a^2 - n 为完全平方数
    a = gmpy2.isqrt(n) + 1
    for i in range(1000000):
        b2 = a * a - n
        b, exact = gmpy2.iroot(int(b2), 2)
        if exact:
            p = int(a + b)
            q = int(a - b)
            if p * q == n:
                print(f"[+] Fermat分解成功: p = {p}, q = {q}")
                phi = (p-1) * (q-1)
                d = int(gmpy2.invert(e, phi))
                m = pow(c, d, n)
                flag = long_to_bytes(m)
                print(f"[+] Flag: {flag}")
                return flag
        a += 1
        
        if i % 100000 == 0:
            print(f"    尝试 {i} 次...")
    
    print("[-] 所有攻击均失败")
    return None

if __name__ == '__main__':
    # 示例：已知n, e, c
    n = 12345678901234567890123456789012345678901234567890  # 示例数字
    e = 65537
    c = 98765432109876543210987654321098765432109876543210
    
    auto_rsa_attack(n, e, c)
```

### 示例2：中国剩余定理在CTF中的应用

```python
#!/usr/bin/env python3
"""CRT在CTF中的多种应用场景"""

from number_theory_tools import NumberTheory
from Crypto.Util.number import *

def scenario1_hastad_broadcast():
    """场景1: Hastad广播攻击 (同一消息被多个公钥加密)"""
    print("=== 场景1: Hastad广播攻击 ===")
    
    # 假设同一消息m被加密3次 (e=3)
    e = 3
    n_list = [
        9797979797979797,  # n1
        9898989898989898,  # n2
        9999999999999999,  # n3
    ]
    c_list = [
        pow(123456789, e, n_list[0]),  # c1
        pow(123456789, e, n_list[1]),  # c2
        pow(123456789, e, n_list[2]),  # c3
    ]
    
    # 使用CRT求解
    nt = NumberTheory()
    x, M = nt.crt(c_list, n_list)
    
    # 开e次根
    m, exact = gmpy2.iroot(x, e)
    print(f"m^e = {x}")
    print(f"m = {m}")
    print(f"exact = {exact}")

def scenario2_multi_prime_rsa():
    """场景2: 多素数RSA (n = p*q*r)"""
    print("\n=== 场景2: 多素数RSA ===")
    
    # 三素数RSA: n = p * q * r
    p, q, r = 61, 53, 47
    n = p * q * r
    phi = (p-1) * (q-1) * (r-1)
    
    e = 17
    d = int(gmpy2.invert(e, phi))
    
    # 加密
    m = 12345
    c = pow(m, e, n)
    
    # 解密 (标准方法)
    m_decrypted = pow(c, d, n)
    print(f"原始: {m}")
    print(f"解密: {m_decrypted}")
    
    # 使用CRT加速解密
    # 分别计算 m_p = c^d mod p, m_q = c^d mod q, m_r = c^d mod r
    dp = d % (p-1)
    dq = d % (q-1)
    dr = d % (r-1)
    
    mp = pow(c, dp, p)
    mq = pow(c, dq, q)
    mr = pow(c, dr, r)
    
    # CRT合并
    nt = NumberTheory()
    m_crt, _ = nt.crt([mp, mq, mr], [p, q, r])
    print(f"CRT解密: {m_crt}")

def scenario3_secret_sharing():
    """场景3: 基于CRT的秘密共享 (Shamir-like)"""
    print("\n=== 场景3: CRT秘密共享 ===")
    
    secret = 12345  # 要分享的秘密
    
    # 选择3个两两互质的模数，都大于secret
    moduli = [127, 131, 137]
    
    # 生成影子
    shares = [secret % m for m in moduli]
    
    print(f"秘密: {secret}")
    print(f"影子: {shares}")
    
    # 恢复秘密（使用全部影子）
    nt = NumberTheory()
    recovered, _ = nt.crt(shares, moduli)
    print(f"恢复: {recovered}")

if __name__ == '__main__':
    scenario1_hastad_broadcast()
    scenario2_multi_prime_rsa()
    scenario3_secret_sharing()
```
